{潜龙出海选股【9:25·定稿】辰星线=白  牵牛线=黄}
{stock-hub: T5突破 + L4量能 + 抗织布}
{主图挂潜龙出海对照  勿用EXPMA}
{宽松对照: 潜龙出海_选股_原版.txt}

M:=1;

YTSL:=(3*CLOSE+LOW+OPEN+HIGH)/6;
辰星线:=(20*YTSL+19*REF(YTSL,1)+18*REF(YTSL,2)+17*REF(YTSL,3)
+16*REF(YTSL,4)+15*REF(YTSL,5)+14*REF(YTSL,6)+13*REF(YTSL,7)
+12*REF(YTSL,8)+11*REF(YTSL,9)+10*REF(YTSL,10)+9*REF(YTSL,11)
+8*REF(YTSL,12)+7*REF(YTSL,13)+6*REF(YTSL,14)+5*REF(YTSL,15)
+4*REF(YTSL,16)+3*REF(YTSL,17)+2*REF(YTSL,18)+REF(YTSL,20))/211;
牵牛线:=MA(CLOSE,26);

{======== T5 昨日突破(形态底线) ========}
昨收:=REF(CLOSE,1);
昨开:=REF(OPEN,1);
昨高:=REF(HIGH,1);
昨低:=REF(LOW,1);
昨白:=REF(辰星线,1);
昨黄:=REF(牵牛线,1);
前收:=REF(CLOSE,2);
前白:=REF(辰星线,2);

当日突破:=CLOSE>辰星线 AND REF(CLOSE,1)<=REF(辰星线,1) AND CLOSE>OPEN;
昨首破白:=昨收>昨白 AND 前收<=前白;
昨阳线:=昨收>昨开;
昨涨幅:=(昨收-前收)/前收*100;
昨站上双线:=昨收>昨白 AND 昨收>昨黄;

形态OK:=REF(当日突破,1) AND BARSLAST(当日突破)=1
AND 昨首破白 AND 昨阳线 AND 昨站上双线
AND 昨涨幅>=2 AND 昨涨幅<=8;

{======== 抗织布(昨收数据) ========}
昨振幅:=(昨高-昨低)/昨收*100;
昨实体:=ABS(昨收-昨开)/昨收*100;
昨收靠上:=昨收>=昨高-(昨高-昨低)*0.35;
近5窄幅:=COUNT((REF(HIGH,1)-REF(LOW,1))/REF(CLOSE,1)*100<3,5);
非织布:=昨振幅>=4 AND 昨实体>=2 AND 昨收靠上 AND 近5窄幅<=2;

{======== L4 量能换手(主筛) ========}
昨量比:=REF(VOL,1)/REF(MA(VOL,5),1);
昨换手:=REF(HSL,1);
五均换手:=REF(MA(HSL,5),1);

量能OK:=昨量比>=1.5 AND 昨量比<=3.5
AND 昨换手>=5 AND 昨换手<=10
AND 昨换手>=五均换手*1.2
AND 昨换手*昨量比>=9
AND COUNT(当日突破,12)<=2;

{======== 9:25 竞价(白线锚定昨日) ========}
平开或高开:=OPEN>=昨收*0.999 AND OPEN<=昨收*1.03;
开盘不破白线:=OPEN>=昨白*1.001;

{======== M=2 尾盘加强(可选) ========}
涨幅合理:=(CLOSE-昨收)/昨收*100;
量比OK:=VOL/MA(VOL,5)>=1.2;
尾盘条件:=涨幅合理>0 AND 涨幅合理<9 AND 量比OK;
可交易:=HIGH>LOW;

XG:形态OK AND 非织布 AND 量能OK AND 平开或高开 AND 开盘不破白线 AND 可交易
AND ((M=1) OR (M=2 AND 尾盘条件));
